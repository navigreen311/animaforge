# GPU / machine-learning extras for the X5 avatar pipeline.
#
# These are NOT installed by CI. The GitHub Actions runner has no CUDA device,
# and the torch + nerfstudio stack is several gigabytes, so `requirements.txt`
# deliberately excludes them and the pytest job installs only that file.
#
# Install these on a CUDA host to enable AVATAR_ENGINE=real:
#
#     pip install -r requirements.txt
#     pip install -r requirements-ml.txt
#
# Then set, in the service environment:
#
#     AVATAR_ENGINE=real
#     AVATAR_WEIGHTS_DIR=/opt/animaforge/weights
#
# Verify with `GET /ai/v1/avatar/capabilities` — it reports the true state of
# every dependency below. Without all of them the real engine refuses to run
# rather than silently returning procedural output.
#
# Hardware floor: NVIDIA GPU with compute capability >= 7.0 and >= 12 GB VRAM
# for splatfacto at default resolution. See docs/avatar-studio.md.

# torch must match the host CUDA runtime. Pick the right wheel index, e.g.
#   pip install torch==2.3.1 --index-url https://download.pytorch.org/whl/cu121
torch==2.3.1

# 3D Gaussian Splatting CUDA rasteriser (Apache-2.0). Builds against torch,
# so install torch first.
gsplat==1.0.0

# nerfstudio provides the `splatfacto` trainer plus the `ns-process-data` and
# `ns-train` CLIs that the real engine shells out to (Apache-2.0).
nerfstudio==1.1.3

# Structure-from-motion for camera pose estimation, used by ns-process-data.
# COLMAP itself is a system package, not a wheel:
#   apt-get install colmap    (or `brew install colmap`)
pycolmap==0.6.1

# Optional: S3 artifact storage (AVATAR_STORAGE_BACKEND=s3).
boto3>=1.34.0


# ---------------------------------------------------------------------------
# Generation clusters beyond X5
# ---------------------------------------------------------------------------
#
# Same rule as above: CI installs requirements.txt only, so nothing here is a
# hard dependency. Each block names the cluster it unlocks and the env var that
# selects it. `GET /ai/v1/capabilities` reports the true state of every one.
#
# IMPORTANT: installing these is necessary but not sufficient. Only the
# clusters marked real-gated below have an engine adapter written. For the
# rest, capabilities reports real_implemented=false and setting the env var to
# `real` fails loudly rather than silently returning synthetic output -- the
# adapter has to be written first. See docs/generation-pipeline.md.

# E6 Continuity (CONTINUITY_ENGINE=real) -- semantic shot similarity.
# The default descriptor is already computed from real pixels on CPU; CLIP adds
# semantic matching the handcrafted descriptor cannot do, e.g. the same actor in
# a different costume.
#   open_clip_torch==2.24.0
#
# X6 Style (STYLE_ENGINE=real) -- style embedding and transfer.
#   open_clip_torch==2.24.0
#
# D3/D6 Video (VIDEO_ENGINE=real) -- text-to-video and image-to-video.
#   diffusers==0.27.2
#   transformers==4.40.0
#   accelerate==0.29.0
#
# D4 Audio (AUDIO_ENGINE=real) -- speech synthesis and forced alignment.
#   torchaudio==2.3.1
#
# G2 Dubbing (DUBBING_ENGINE=real) -- multi-language dubbing, voice cloning.
#   torchaudio==2.3.1
#
# D10 Training (TRAINING_ENGINE=real) -- LoRA / DreamBooth fine-tuning.
#   peft==0.10.0
#
# F3 Music (MUSIC_ENGINE=real) -- music generation and adaptive scoring.
#   audiocraft==1.3.0
#
# G6 Auto-QC needs nothing here: loudness, temporal stability, A/V sync and
# container validation are numpy on CPU. ffmpeg on PATH widens what can be
# decoded (currently WAV for audio, PNG for frames) but is not required -- what
# cannot be decoded is reported unmeasurable rather than scored.

# ---------------------------------------------------------------------------
# Generation clusters (D3/D6 video, D4 audio, X6 style, F3 music, D10 training)
# ---------------------------------------------------------------------------
#
# Same rule as above: none of this is installed by CI, and none of it may
# become a transitive dependency of requirements.txt. Each cluster names the
# packages it needs in the REGISTRY in src/services/engines.py, and
# GET /ai/v1/capabilities reports which are actually importable here.
#
# Per-cluster hardware floors are in docs/generation-pipeline.md.

# D4 audio -- CTC forced alignment. CPU is fine; this is the cheapest upgrade
# in the set. Must match the torch version above.
torchaudio==2.3.1

# X6 style -- CLIP image embeddings. Inference is CPU-feasible for ViT-B/32.
open_clip_torch==2.24.0

# D3/D6 video and D10 training -- diffusion pipelines and LoRA adapters.
# Both need a CUDA device; see the doc for VRAM.
diffusers==0.27.2
transformers==4.40.2
accelerate==0.30.1
peft==0.10.0
safetensors==0.4.3

# Video export muxes frames through imageio-ffmpeg, which ships its own
# ffmpeg binary, so no system package is required for the encode step.
imageio-ffmpeg==0.4.9

# Image loading for the CLIP, image-to-video and LoRA paths.
Pillow==10.3.0

# F3 music -- MusicGen. Pulls its own torch pin; install after torch so the
# resolver does not pick a different build.
audiocraft==1.3.0
